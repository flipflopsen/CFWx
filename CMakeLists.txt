cmake_minimum_required(VERSION 3.16)
project(CFWx VERSION 0.1.0 LANGUAGES C)
# Optional vcpkg: pass CMAKE_TOOLCHAIN_FILE externally on first configure.
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
option(CFWX_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(cfwx_core STATIC
    src/datastructures/vector.c src/datastructures/matrix.c
    src/datastructures/tensor.c src/datastructures/neuron.c
    src/maths/activation_functions.c src/maths/loss_functions.c
    src/maths/rewards.c src/models/ann.c src/models/attention.c)
target_include_directories(cfwx_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(NOT MSVC)
    target_link_libraries(cfwx_core PUBLIC m)
endif()
add_executable(cfwx src/main.c src/util/args.c src/util/logger.c src/util/messages.c)
target_link_libraries(cfwx PRIVATE cfwx_core)
target_compile_definitions(cfwx PRIVATE PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(cfwx_test_runner test/main.c)
    target_link_libraries(cfwx_test_runner PRIVATE cfwx_core)
    add_test(NAME numerical_foundations COMMAND cfwx_test_runner)
endif()
foreach(target cfwx_core cfwx cfwx_test_runner)
    if(TARGET ${target} AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
        if(CFWX_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
find_program(VALGRIND_EXECUTABLE valgrind)
if(VALGRIND_EXECUTABLE AND BUILD_TESTING)
    add_custom_target(valgrind
        COMMAND ${VALGRIND_EXECUTABLE} --leak-check=full --error-exitcode=1 $<TARGET_FILE:cfwx_test_runner>
        DEPENDS cfwx_test_runner VERBATIM)
endif()
